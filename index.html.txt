<script>
  const firebaseConfig = {
    apiKey: "AIzaSyAyn6xM-pkDVK7ojbQXCjfHO07P8rwevMQ",
    authDomain: "burncreekspeedshop.firebaseapp.com",
    projectId: "burncreekspeedshop",
    storageBucket: "burncreekspeedshop.firebasestorage.app",
    messagingSenderId: "611870239128",
    appId: "1:611870239128:web:5e8973ec49556be0f823ed"
  };

  firebase.initializeApp(firebaseConfig);
  const auth = firebase.auth();
  const db = firebase.firestore();
  const storage = firebase.storage();

  auth.onAuthStateChanged(user => {
    if (user) {
      document.getElementById('auth-view').style.display = 'none';
      document.getElementById('main-view').style.display = 'flex';
      document.getElementById('user-display').innerText = `FÖRARE: ${user.displayName || user.email.split('@')[0]}`;
      loadFeed();
      loadTodos();
    } else {
      document.getElementById('auth-view').style.display = 'block';
      document.getElementById('main-view').style.display = 'none';
    }
  });

  async function register() {
    const name = document.getElementById('auth-name').value.trim();
    const email = document.getElementById('auth-email').value.trim();
    const pass = document.getElementById('auth-password').value;
    if (!name) return alert('Ange förarnamn!');
    try {
      const res = await auth.createUserWithEmailAndPassword(email, pass);
      await res.user.updateProfile({ displayName: name });
      location.reload();
    } catch (err) { document.getElementById('auth-error').innerText = err.message; }
  }

  async function login() {
    const email = document.getElementById('auth-email').value.trim();
    const pass = document.getElementById('auth-password').value;
    try {
      await auth.signInWithEmailAndPassword(email, pass);
    } catch (err) { document.getElementById('auth-error').innerText = err.message; }
  }

  function logout() { auth.signOut(); }

  function switchTab(tab) {
    document.querySelectorAll('.tab-btn').forEach((b, i) => b.classList.toggle('active', (i === 0 && tab === 'feed') || (i === 1 && tab === 'todo')));
    document.getElementById('feed-tab').classList.toggle('active', tab === 'feed');
    document.getElementById('todo-tab').classList.toggle('active', tab === 'todo');
  }

  function updateFileName() {
    const f = document.getElementById('post-file').files[0];
    document.getElementById('file-chosen').innerText = f ? f.name : '';
  }

  async function publishPost() {
    const text = document.getElementById('post-text').value.trim();
    const file = document.getElementById('post-file').files[0];
    if (!text && !file) return;

    let imageUrl = '';
    if (file) {
      const ref = storage.ref(`posts/${Date.now()}_${file.name}`);
      const upload = await ref.put(file);
      imageUrl = await upload.ref.getDownloadURL();
    }

    await db.collection('posts').add({
      text: text,
      imageUrl: imageUrl,
      author: auth.currentUser.displayName || auth.currentUser.email.split('@')[0],
      createdAt: firebase.firestore.FieldValue.serverTimestamp()
    });

    document.getElementById('post-text').value = '';
    document.getElementById('post-file').value = '';
    document.getElementById('file-chosen').innerText = '';
  }

  function loadFeed() {
    db.collection('posts').orderBy('createdAt', 'desc').limit(25).onSnapshot(snapshot => {
      const container = document.getElementById('posts-container');
      container.innerHTML = '';
      snapshot.forEach(doc => {
        const p = doc.data();
        const date = p.createdAt ? new Date(p.createdAt.toDate()).toLocaleDateString('sv-SE', { hour: '2-digit', minute: '2-digit' }) : 'Nyss';
        container.innerHTML += `
          <div class="post-card">
            <div class="post-header">
              <span class="post-author">${p.author}</span>
              <span>${date}</span>
            </div>
            ${p.text ? `<p style="font-size: 0.95rem; line-height: 1.4;">${p.text}</p>` : ''}
            ${p.imageUrl ? `<img src="${p.imageUrl}" class="post-image" loading="lazy">` : ''}
          </div>
        `;
      });
    });
  }

  async function addTodo() {
    const input = document.getElementById('todo-input');
    const val = input.value.trim();
    if (!val) return;
    await db.collection('todos').add({ text: val, done: false, addedBy: auth.currentUser.displayName || 'Depå' });
    input.value = '';
  }

  function toggleTodo(id, currentStatus) {
    db.collection('todos').doc(id).update({ done: !currentStatus });
  }

  function loadTodos() {
    db.collection('todos').onSnapshot(snapshot => {
      const container = document.getElementById('todos-container');
      container.innerHTML = '';
      snapshot.forEach(doc => {
        const t = doc.data();
        container.innerHTML += `
          <div class="todo-item ${t.done ? 'done' : ''}" onclick="toggleTodo('${doc.id}', ${t.done})">
            <input type="checkbox" ${t.done ? 'checked' : ''} style="cursor: pointer; accent-color: var(--fire-orange);">
            <span style="flex: 1;">${t.text}</span>
            <span class="todo-badge">${t.addedBy}</span>
          </div>
        `;
      });
    });
  }

  if ('serviceWorker' in navigator) {
    navigator.serviceWorker.register('sw.js');
  }
